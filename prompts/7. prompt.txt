```javascript
const { remote } = require('webdriverio');

let driver;

async function getDriver() {
  if (driver) {
    return driver;
  }
  const capabilities = {
    platformName: 'Android',
    'appium:automationName': 'UiAutomator2',
    'appium:deviceName': 'emulator-5554',
    'appium:app': '../health_pal_frontend/build/app/outputs/flutter-apk/app-release.apk',
  };
  driver = await remote({
    path: '/wd/hub',
    port: 4723,
    capabilities,
    logLevel: 'error',
  });
  return driver;
}

module.exports = { getDriver };
```
