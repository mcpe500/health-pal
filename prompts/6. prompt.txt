```javascript
const axios = require('axios');
const { API_BASE_URL, DUMMY_GOOGLE_ID_TOKEN } = require('./config');

describe('Auth API', () => {
  let jwtToken;

  test('should login with a new user and return a JWT', async () => {
    const response = await axios.post(`${API_BASE_URL}/auth/google-login`, {
      id_token: DUMMY_GOOGLE_ID_TOKEN,
    });
    expect(response.status).toBe(200);
    expect(response.data.token).toBeDefined();
    jwtToken = response.data.token;
  });

  test('should access a protected route with a valid JWT', async () => {
    const response = await axios.get(`${API_BASE_URL}/api/v1/profile`, {
      headers: { Authorization: `Bearer ${jwtToken}` },
    });
    expect(response.status).toBe(200);
    expect(response.data.email).toBeDefined();
  });

  test('should fail to access protected route with an invalid JWT', async () => {
    try {
      await axios.get(`${API_BASE_URL}/api/v1/profile`, {
        headers: { Authorization: 'Bearer invalidtoken' },
      });
    } catch (error) {
      expect(error.response.status).toBe(401);
    }
  });
});
```
